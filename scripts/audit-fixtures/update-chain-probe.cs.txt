var findings = new List<object>();
var fixtureRoot = Path.Combine(Path.GetTempPath(), "LanDesktopUpdateAudit", Guid.NewGuid().ToString("N"));
Directory.CreateDirectory(fixtureRoot);
var source = new PlondsSourceDescriptor("fixture", "s3", "https://audit.test/PLONDS.json", 100);
PlondsClientManifest Manifest(string version = "2.0.0", string previous = "1.5.0", string channel = "stable", string platform = "windows-x64") =>
    new("2.0", version, previous, false, false, channel, platform, DateTimeOffset.UtcNow, new Dictionary<string, PlondsClientFileEntry>(), new Dictionary<string, PlondsClientChangedFileEntry>(), new Dictionary<string, string>(), null, null);
string Hash(string text) => Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(text))).ToLowerInvariant();

// Discovery accepts a higher version from another architecture and channel.
var selected = PlondsManifestSelector.SelectHighestVersionCandidates([
    new(source, Manifest("2.0.0", channel: "stable", platform: "windows-x64")),
    new(source with { Id = "preview" }, Manifest("3.0.0", channel: "preview", platform: "windows-x64")),
    new(source with { Id = "x86" }, Manifest("4.0.0", channel: "stable", platform: "windows-x86"))
], "windows-x64", "stable");
findings.Add(new { id = "selection", requested = "stable/windows-x64", chosen = selected[0].Manifest.CurrentVersion, selected[0].Manifest.Channel, selected[0].Manifest.Platform });

// Use the real discovery and planner with a recorded downloader, rather than duplicating selection code.
var recordingDownloader = new RecordingDownloader();
using (var http = new HttpClient(new FixtureHandler(JsonSerializer.Serialize(Manifest()), [])))
{
    var plonds = new PlondsService(new PlondsSourceRegistry([source]), new PlondsManifestClient(http), new PlondsDownloadPlanner(recordingDownloader));
    var prepared = await plonds.FindAndPrepareLatestAsync(new Version(1, 0, 0), CancellationToken.None);
    findings.Add(new { id = "baseline", installed = "1.0.0", manifestBaseline = "1.5.0", prepared.Success, recordingDownloader.DeltaCalls, recordingDownloader.FullCalls });
}

// A clean-install full package includes changed binaries outside app-*.
var hostRoot = Path.Combine(fixtureRoot, "host-update");
var oldApp = Path.Combine(hostRoot, "app-1.0.0");
Directory.CreateDirectory(oldApp);
File.WriteAllText(Path.Combine(oldApp, "LanMountainDesktop.exe"), "old-host");
File.WriteAllText(Path.Combine(oldApp, ".current"), "");
File.WriteAllText(Path.Combine(hostRoot, "LanMountainDesktop.Launcher.exe"), "old-launcher");
File.WriteAllText(Path.Combine(hostRoot, "LanMountainDesktop.AirAppRuntime.exe"), "old-runtime");
var filesRoot = Path.Combine(fixtureRoot, "full-package");
var nextApp = Path.Combine(filesRoot, "app-2.0.0");
Directory.CreateDirectory(nextApp);
File.WriteAllText(Path.Combine(nextApp, "LanMountainDesktop.exe"), "corrupt-host");
File.WriteAllText(Path.Combine(filesRoot, "LanMountainDesktop.Launcher.exe"), "new-launcher");
File.WriteAllText(Path.Combine(filesRoot, "LanMountainDesktop.AirAppRuntime.exe"), "new-runtime");
var manifestPath = Path.Combine(fixtureRoot, "PLONDS.json");
var fullManifest = Manifest() with { RequiresCleanInstall = true, FilesMap = new Dictionary<string, PlondsClientFileEntry> { ["LanMountainDesktop.exe"] = new("replace", Hash("expected-host"), 13) } };
File.WriteAllText(manifestPath, JsonSerializer.Serialize(fullManifest));
var fullPackage = new PlondsPreparedPackage(new Version(2, 0, 0), PlondsPackageMode.Full, manifestPath, null, null, null, filesRoot);
var hostInstall = await new PlondsPreparedPackageInstaller().InstallAsync(fullPackage, hostRoot, null, CancellationToken.None);
var activeApp = Directory.GetDirectories(hostRoot, "app-*").Single(p => File.Exists(Path.Combine(p, ".current")));
var rollback = new UpdateRollbackGateway().RollbackLatest(hostRoot);
findings.Add(new {
    id = "host-full-install", hostInstall.Success,
    launcher = File.ReadAllText(Path.Combine(hostRoot, "LanMountainDesktop.Launcher.exe")),
    runtime = File.ReadAllText(Path.Combine(hostRoot, "LanMountainDesktop.AirAppRuntime.exe")),
    deployedHost = File.ReadAllText(Path.Combine(activeApp, "LanMountainDesktop.exe")),
    actualHashMatchesManifest = Hash(File.ReadAllText(Path.Combine(activeApp, "LanMountainDesktop.exe"))) == fullManifest.FilesMap["LanMountainDesktop.exe"].Hash,
    oldDeploymentMarkedDestroy = File.Exists(Path.Combine(oldApp, ".destroy")),
    rollbackSuccess = rollback.Success, rollback.Code
});

// Cancellation changes the phase while the real DownloadPlondsAsync continuation remains live.
var settings = new MemorySettings();
var waiting = new WaitingPlonds(fullPackage, Manifest(previous: "1.0.0"));
using (var update = new UpdateSettingsService(settings, () => throw new InvalidOperationException("unused"), waiting))
{
    await update.CheckAsync();
    var downloadTask = update.DownloadAsync();
    await waiting.Entered.Task;
    await update.PauseAsync();
    var phaseAfterPause = update.CurrentPhase.ToString();
    var cancellationAfterPause = waiting.DownloadToken.IsCancellationRequested;
    await update.CancelAsync();
    var phaseAfterCancel = update.CurrentPhase.ToString();
    var cancellationAfterCancel = waiting.DownloadToken.IsCancellationRequested;
    waiting.Completion.SetResult(PlondsPrepareResult.Prepared(fullPackage));
    var download = await downloadTask;
    findings.Add(new { id = "cancel", phaseAfterPause, cancellationAfterPause, phaseAfterCancel, cancellationAfterCancel, download.Success, finalPhase = update.CurrentPhase.ToString(), settings.Snapshot.PendingUpdateInstallerPath });
}
using (var restarted = new UpdateSettingsService(settings, () => throw new InvalidOperationException("unused"), waiting))
    findings.Add(new { id = "pending-after-restart", savedPath = settings.Snapshot.PendingUpdateInstallerPath, phase = restarted.CurrentPhase.ToString() });

// Real online incremental install consumes the same nested root layout produced by release.yml.
var onlineRoot = Path.Combine(fixtureRoot, "online-install");
var onlineOld = Path.Combine(onlineRoot, "app-1.0.0");
Directory.CreateDirectory(onlineOld);
File.WriteAllText(Path.Combine(onlineOld, ".current"), "");
File.WriteAllText(Path.Combine(onlineOld, "LanMountainDesktop.exe"), "old-host");
var zipPath = Path.Combine(fixtureRoot, "Files.zip");
using (var zip = ZipFile.Open(zipPath, ZipArchiveMode.Create))
{
    using var writer = new StreamWriter(zip.CreateEntry("app-2.0.0/LanMountainDesktop.exe").Open());
    writer.Write("new-host");
}
var zipBytes = File.ReadAllBytes(zipPath);
var onlineManifest = new InstallerPlondsManifest("2.0", "2.0.0", "1.0.0", false, true, "preview", "windows-x64", DateTimeOffset.UtcNow,
    new Dictionary<string, InstallerPlondsFileEntry> { ["LanMountainDesktop.exe"] = new("replace", Hash("new-host"), 8) },
    new Dictionary<string, InstallerPlondsChangedFileEntry>(), new Dictionary<string, string> { ["Files.zip"] = "sha256:" + Convert.ToHexString(SHA256.HashData(zipBytes)).ToLowerInvariant() }, null, null);
using (var http = new HttpClient(new FixtureHandler(JsonSerializer.Serialize(onlineManifest), zipBytes)))
{
    var client = new InstallerPlondsClient(http, Path.Combine(fixtureRoot, "online-staging"), _ => TimeSpan.Zero, "preview");
    var service = new OnlineInstallService(client, new FilesPackageInstaller((_, _) => { }), new FixedPrivacy(), new InstalledProductInspector(), new IncrementalPlanBuilder());
    await service.CheckLatestAsync(CancellationToken.None);
    await service.UpdateIncrementalAsync(onlineRoot, null, CancellationToken.None);
    var onlineActive = Directory.GetDirectories(onlineRoot, "app-*").Single(p => File.Exists(Path.Combine(p, ".current")));
    findings.Add(new { id = "online-incremental", returnedSuccessfully = true, onlineManifest.RequiresCleanInstall, activeDirectory = onlineActive, activeHostExists = File.Exists(Path.Combine(onlineActive, "LanMountainDesktop.exe")), oldCurrentStillExists = File.Exists(Path.Combine(onlineOld, ".current")) });
}
var result = JsonSerializer.Serialize(new { fixtureRoot, findings }, new JsonSerializerOptions { WriteIndented = true });
if (args.Length > 0) File.WriteAllText(args[0], result);
Console.WriteLine(result);

sealed class FixtureHandler(string manifestJson, byte[] zipBytes) : HttpMessageHandler
{
    protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken) =>
        Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK) {
            Content = request.RequestUri!.AbsolutePath.EndsWith("PLONDS.json", StringComparison.OrdinalIgnoreCase)
                ? new StringContent(manifestJson, Encoding.UTF8, "application/json")
                : new ByteArrayContent(zipBytes)
        });
}
sealed class RecordingDownloader : IPlondsPackageDownloader
{
    public int DeltaCalls { get; private set; }
    public int FullCalls { get; private set; }
    public Task<PlondsPreparedPackage> PrepareDeltaAsync(PlondsClientManifest manifest, PlondsSourceDescriptor source, CancellationToken ct) { DeltaCalls++; return Task.FromResult(new PlondsPreparedPackage(Version.Parse(manifest.CurrentVersion), PlondsPackageMode.Delta, "fixture", null, null, null, null)); }
    public Task<PlondsPreparedPackage> PrepareFullAsync(PlondsClientManifest manifest, PlondsSourceDescriptor source, CancellationToken ct) { FullCalls++; return Task.FromResult(new PlondsPreparedPackage(Version.Parse(manifest.CurrentVersion), PlondsPackageMode.Full, "fixture", null, null, null, null)); }
}
sealed class WaitingPlonds(PlondsPreparedPackage package, PlondsClientManifest manifest) : IPlondsService
{
    public TaskCompletionSource Entered { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
    public TaskCompletionSource<PlondsPrepareResult> Completion { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
    public CancellationToken DownloadToken { get; private set; }
    public Task<PlondsLatestResult> FindLatestAsync(Version current, CancellationToken ct) => Task.FromResult(PlondsLatestResult.Available(current, package.Version, [new(new("fixture", "s3", "https://audit.test/PLONDS.json", 100), manifest)]));
    public Task<PlondsPrepareResult> FindAndPrepareLatestAsync(CancellationToken ct) => FindAndPrepareLatestAsync(new Version(1, 0, 0), false, ct);
    public Task<PlondsPrepareResult> FindAndPrepareLatestAsync(Version current, CancellationToken ct) => FindAndPrepareLatestAsync(current, false, ct);
    public Task<PlondsPrepareResult> FindAndPrepareLatestAsync(Version current, bool forceFullPackage, CancellationToken ct) { DownloadToken = ct; Entered.TrySetResult(); return Completion.Task.WaitAsync(ct); }
}
sealed class FixedPrivacy : IPrivacyDeviceIdentityProvider { public string GetOrCreateDeviceId() => "audit-fixture"; }
sealed class MemorySettings : ISettingsService
{
    public AppSettingsSnapshot Snapshot { get; private set; } = new();
    public event EventHandler<SettingsChangedEvent>? Changed;
    public T LoadSnapshot<T>(AirAppSettingsScope scope, string? subjectId = null, string? placementId = null) where T : new() => typeof(T) == typeof(AppSettingsSnapshot) ? (T)(object)Snapshot.Clone() : new();
    public void SaveSnapshot<T>(AirAppSettingsScope scope, T snapshot, string? subjectId = null, string? placementId = null, string? sectionId = null, IReadOnlyCollection<string>? changedKeys = null) { if (snapshot is AppSettingsSnapshot app) Snapshot = app.Clone(); Changed?.Invoke(this, new(scope, subjectId, placementId, sectionId, changedKeys)); }
    public T LoadSection<T>(AirAppSettingsScope scope, string subjectId, string sectionId, string? placementId = null) where T : new() => new();
    public void SaveSection<T>(AirAppSettingsScope scope, string subjectId, string sectionId, T section, string? placementId = null, IReadOnlyCollection<string>? changedKeys = null) => throw new NotSupportedException();
    public void DeleteSection(AirAppSettingsScope scope, string subjectId, string sectionId, string? placementId = null) => throw new NotSupportedException();
    public T? GetValue<T>(AirAppSettingsScope scope, string key, string? subjectId = null, string? placementId = null, string? sectionId = null) => default;
    public void SetValue<T>(AirAppSettingsScope scope, string key, T value, string? subjectId = null, string? placementId = null, string? sectionId = null, IReadOnlyCollection<string>? changedKeys = null) => throw new NotSupportedException();
    public IComponentSettingsAccessor GetComponentAccessor(string componentId, string? placementId) => throw new NotSupportedException();
}
